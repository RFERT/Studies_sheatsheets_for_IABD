⚙️ Générateurs & décorateurs📘 Module 0

def compte(n):
    i = 0
    while i < n:
        yield i      # lazy
        i += 1

import functools
def deco(fn):
    @functools.wraps(fn)
    def w(*a, **k):
        return fn(*a, **k)
    return w

@functools.lru_cache
def fib(n): return n if n<2 else fib(n-1)+fib(n-2)
